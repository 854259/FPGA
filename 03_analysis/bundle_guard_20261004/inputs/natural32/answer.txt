module TopModule (
    input  [31:0] A,
    input  [31:0] B,
    output [31:0] S,
    output        C32
);

    // Internal wires for carry chain between 16-bit CLA blocks
    wire        c16;
    wire [15:0] S_lo;
    wire [15:0] S_hi;

    // Lower 16-bit CLA block (bits 0-15)
    cla_16bit cla_lo (
        .A(A[15:0]),
        .B(B[15:0]),
        .Cin(1'b0),
        .S(S_lo),
        .Cout(c16)
    );

    // Upper 16-bit CLA block (bits 16-31)
    cla_16bit cla_hi (
        .A(A[31:16]),
        .B(B[31:16]),
        .Cin(c16),
        .S(S_hi),
        .Cout(C32)
    );

    // Concatenate sum outputs
    assign S = {S_hi, S_lo};

endmodule

// 16-bit Carry-Lookahead Adder Block
module cla_16bit (
    input  [15:0] A,
    input  [15:0] B,
    input         Cin,
    output [15:0] S,
    output        Cout
);

    // Generate and Propagate signals for each bit
    wire [15:0] G; // Generate: G[i] = A[i] & B[i]
    wire [15:0] P; // Propagate: P[i] = A[i] ^ B[i]

    // Carry signals
    wire [15:0] C; // C[0] is Cin, C[16] is Cout

    assign C[0] = Cin;

    // Generate G and P
    genvar i;
    generate
        for (i = 0; i < 16; i = i + 1) begin : gen_gp
            assign G[i] = A[i] & B[i];
            assign P[i] = A[i] ^ B[i];
        end
    endgenerate

    // Carry-Lookahead Logic for 16 bits
    // C[i+1] = G[i] | (P[i] & C[i])
    // Unrolled for synthesis clarity and speed
    assign C[1]  = G[0]  | (P[0]  & C[0]);
    assign C[2]  = G[1]  | (P[1]  & C[1]);
    assign C[3]  = G[2]  | (P[2]  & C[2]);
    assign C[4]  = G[3]  | (P[3]  & C[3]);
    assign C[5]  = G[4]  | (P[4]  & C[4]);
    assign C[6]  = G[5]  | (P[5]  & C[5]);
    assign C[7]  = G[6]  | (P[6]  & C[6]);
    assign C[8]  = G[7]  | (P[7]  & C[7]);
    assign C[9]  = G[8]  | (P[8]  & C[8]);
    assign C[10] = G[9]  | (P[9]  & C[9]);
    assign C[11] = G[10] | (P[10] & C[10]);
    assign C[12] = G[11] | (P[11] & C[11]);
    assign C[13] = G[12] | (P[12] & C[12]);
    assign C[14] = G[13] | (P[13] & C[13]);
    assign C[15] = G[14] | (P[14] & C[14]);
    assign C[16] = G[15] | (P[15] & C[15]);

    assign Cout = C[16];

    // Sum calculation
    generate
        for (i = 0; i < 16; i = i + 1) begin : gen_sum
            assign S[i] = P[i] ^ C[i];
        end
    endgenerate

endmodule
