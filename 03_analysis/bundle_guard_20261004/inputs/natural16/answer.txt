module TopModule (
    input  [15:0] a,
    input  [15:0] b,
    input         Cin,
    output [15:0] y,
    output        Co
);

    wire [7:0]  sum_lo;
    wire        carry_mid;
    wire [7:0]  sum_hi;

    // Lower 8-bit adder
    adder_8bit adder_lo (
        .a   (a[7:0]),
        .b   (b[7:0]),
        .Cin (Cin),
        .y   (sum_lo),
        .Co  (carry_mid)
    );

    // Upper 8-bit adder
    adder_8bit adder_hi (
        .a   (a[15:8]),
        .b   (b[15:8]),
        .Cin (carry_mid),
        .y   (sum_hi),
        .Co  (Co)
    );

    assign y = {sum_hi, sum_lo};

endmodule

module adder_8bit (
    input  [7:0] a,
    input  [7:0] b,
    input        Cin,
    output [7:0] y,
    output       Co
);

    wire [8:0] result;

    assign result = a + b + Cin;
    assign y = result[7:0];
    assign Co = result[8];

endmodule
